12impl NsProof {
13    /// Returns the payload bytes for the `index`th namespace, along with a
14    /// proof of correctness for those bytes. Returns `None` on error.
15    ///
16    /// The namespace payload [`NsPayloadOwned`] is included as a hidden field
17    /// in the returned [`NsProof`]. A conventional API would instead return
18    /// `(NsPayload, NsProof)` and [`NsProof`] would not contain the namespace
19    /// payload.
20    /// ([`TxProof::new`](crate::block::namespace_payload::TxProof::new)
21    /// conforms to this convention.) In the future we should change this API to
22    /// conform to convention. But that would require a change to our RPC
23    /// endpoint API at [`endpoints`](crate::api::endpoints), which is a hassle.
24    pub fn new(payload: &Payload, index: &NsIndex, common: &VidCommon) -> Option<NsProof> {
25        let payload_byte_len = payload.byte_len();
26        if !payload_byte_len.is_consistent(common) {
27            tracing::warn!(
28                "payload byte len {} inconsistent with common {}",
29                payload_byte_len,
30                VidSchemeType::get_payload_byte_len(common)
31            );
32            return None; // error: payload byte len inconsistent with common
33        }
34        if !payload.ns_table().in_bounds(index) {
35            tracing::warn!("ns_index {:?} out of bounds", index);
36            return None; // error: index out of bounds
37        }
38        let ns_payload_range = payload.ns_table().ns_range(index, &payload_byte_len);
39
40        // TODO vid_scheme() arg should be u32 to match get_num_storage_nodes
41        // https://github.com/EspressoSystems/HotShot/issues/3298
42        let vid = vid_scheme(
43            VidSchemeType::get_num_storage_nodes(common)
44                .try_into()
45                .ok()?, // error: failure to convert u32 to usize
46        );
47
48        let ns_proof = if ns_payload_range.as_block_range().is_empty() {
49            None
50        } else {
51            Some(
52                vid.payload_proof(payload.encode(), ns_payload_range.as_block_range())
53                    .ok()?, // error: internal to payload_proof()
54            )
55        };
56
57        Some(NsProof {
58            ns_index: index.clone(),
59            ns_payload: payload.read_ns_payload(&ns_payload_range).to_owned(),
60            ns_proof,
61        })
62    }
63
64    /// Verify a [`NsProof`] against a payload commitment. Returns `None` on
65    /// error or if verification fails.
66    ///
67    /// There is no [`NsPayload`](crate::block::namespace_payload::NsPayload)
68    /// arg because this data is already included in the [`NsProof`]. See
69    /// [`NsProof::new`] for discussion.
70    ///
71    /// If verification is successful then return `(Vec<Transaction>,
72    /// NamespaceId)` obtained by post-processing the underlying
73    /// [`NsPayload`](crate::block::namespace_payload::NsPayload). Why? This
74    /// method might be run by a client in a WASM environment who might be
75    /// running non-Rust code, in which case the client is unable to perform
76    /// this post-processing himself.
77    pub fn verify(
78        &self,
79        ns_table: &NsTable,
80        commit: &VidCommitment,
81        common: &VidCommon,
82    ) -> Option<(Vec<Transaction>, NamespaceId)> {
83        VidSchemeType::is_consistent(commit, common).ok()?;
84        if !ns_table.in_bounds(&self.ns_index) {
85            return None; // error: index out of bounds
86        }
87
88        let range = ns_table
89            .ns_range(&self.ns_index, &PayloadByteLen::from_vid_common(common))
90            .as_block_range();
91
92        match (&self.ns_proof, range.is_empty()) {
93            (Some(proof), false) => {
94                // TODO vid_scheme() arg should be u32 to match get_num_storage_nodes
95                // https://github.com/EspressoSystems/HotShot/issues/3298
96                let vid = vid_scheme(
97                    VidSchemeType::get_num_storage_nodes(common)
98                        .try_into()
99                        .ok()?, // error: failure to convert u32 to usize
100                );
101
102                vid.payload_verify(
103                    Statement {
104                        payload_subslice: self.ns_payload.as_bytes_slice(),
105                        range,
106                        commit,
107                        common,
108                    },
109                    proof,
110                )
111                .ok()? // error: internal to payload_verify()
112                .ok()?; // verification failure
113            }
114            (None, true) => {} // 0-length namespace, nothing to verify
115            (None, false) => {
116                tracing::error!(
117                    "ns verify: missing proof for nonempty ns payload range {:?}",
118                    range
119                );
120                return None;
121            }
122            (Some(_), true) => {
123                tracing::error!("ns verify: unexpected proof for empty ns payload range");
124                return None;
125            }
126        }
127
128        // verification succeeded, return some data
129        let ns_id = ns_table.read_ns_id_unchecked(&self.ns_index);
130        Some((self.ns_payload.export_all_txs(&ns_id), ns_id))
131    }
132
133    /// Return all transactions in the namespace whose payload is proven by
134    /// `self`. The namespace ID for each returned [`Transaction`] is set to
135    /// `ns_id`.
136    ///
137    /// # Design warning
138    ///
139    /// This method relies on a promise that a [`NsProof`] stores the entire
140    /// namespace payload. If in the future we wish to remove the payload from a
141    /// [`NsProof`] then this method can no longer be supported.
142    ///
143    /// In that case, use the following a workaround:
144    /// - Given a [`NamespaceId`], get a [`NsIndex`] `i` via
145    ///   [`NsTable::find_ns_id`].
146    /// - Use `i` to get a
147    ///   [`NsPayload`](crate::block::namespace_payload::NsPayload) `p` via
148    ///   [`Payload::ns_payload`].
149    /// - Use `p` to get the desired [`Vec<Transaction>`] via
150    ///   [`NsPayload::export_all_txs`](crate::block::namespace_payload::NsPayload::export_all_txs).
151    ///
152    /// This workaround duplicates the work done in [`NsProof::new`]. If you
153    /// don't like that then you could instead hack [`NsProof::new`] to return a
154    /// pair `(NsProof, Vec<Transaction>)`.
155    pub fn export_all_txs(&self, ns_id: &NamespaceId) -> Vec<Transaction> {
156        self.ns_payload.export_all_txs(ns_id)
157    }
158}
