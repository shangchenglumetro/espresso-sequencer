50pub(super) fn get_balance<State, Ver>() -> Result<Api<State, merklized_state::Error, Ver>>
51where
52    State: 'static + Send + Sync + ReadState,
53    Ver: 'static + StaticVersionType,
54    <State as ReadState>::State: Send
55        + Sync
56        + MerklizedStateDataSource<SeqTypes, FeeMerkleTree, { FeeMerkleTree::ARITY }>
57        + MerklizedStateHeightPersistence,
58{
59    let mut options = merklized_state::Options::default();
60    let extension = toml::from_str(include_str!("../../api/merklized_state.toml"))?;
61    options.extensions.push(extension);
62
63    let mut api =
64        merklized_state::define_api::<State, SeqTypes, FeeMerkleTree, Ver, 256>(&options)?;
65
66    api.get("getfeebalance", move |req, state| {
67        async move {
68            let address = req.string_param("address")?;
69            let height = state.get_last_state_height().await?;
70            let snapshot = Snapshot::Index(height as u64);
71            let key = address
72                .parse()
73                .map_err(|_| merklized_state::Error::Custom {
74                    message: "failed to parse address".to_string(),
75                    status: StatusCode::BAD_REQUEST,
76                })?;
77            let path = state.get_path(snapshot, key).await?;
78            Ok(path.elem().copied())
79        }
80        .boxed()
81    })?;
82    Ok(api)
83}

92pub(super) fn availability<N, P, D, V: Versions>(
93) -> Result<AvailabilityApi<N, P, D, V, SequencerApiVersion>>
94where
95    N: ConnectedNetwork<PubKey>,
96    D: SequencerDataSource + Send + Sync + 'static,
97    P: SequencerPersistence,
98{
99    let mut options = availability::Options::default();
100    let extension = toml::from_str(include_str!("../../api/availability.toml"))?;
101    options.extensions.push(extension);
102    let timeout = options.fetch_timeout;
103
104    let mut api = availability::define_api::<AvailState<N, P, D, _>, SeqTypes, _>(
105        &options,
106        SequencerApiVersion::instance(),
107    )?;
108
109    api.get("getnamespaceproof", move |req, state| {
110        async move {
111            let height: usize = req.integer_param("height")?;
112            let ns_id = NamespaceId::from(req.integer_param::<_, u32>("namespace")?);
113            let (block, common) = try_join!(
114                async move {
115                    state
116                        .get_block(height)
117                        .await
118                        .with_timeout(timeout)
119                        .await
120                        .context(FetchBlockSnafu {
121                            resource: height.to_string(),
122                        })
123                },
124                async move {
125                    state
126                        .get_vid_common(height)
127                        .await
128                        .with_timeout(timeout)
129                        .await
130                        .context(FetchBlockSnafu {
131                            resource: height.to_string(),
132                        })
133                }
134            )?;
135
136            if let Some(ns_index) = block.payload().ns_table().find_ns_id(&ns_id) {
137                let proof = NsProof::new(block.payload(), &ns_index, common.common()).context(
138                    CustomSnafu {
139                        message: format!("failed to make proof for namespace {ns_id}"),
140                        status: StatusCode::NOT_FOUND,
141                    },
142                )?;
143
144                Ok(NamespaceProofQueryData {
145                    transactions: proof.export_all_txs(&ns_id),
146                    proof: Some(proof),
147                })
148            } else {
149                // ns_id not found in ns_table
150                Ok(NamespaceProofQueryData {
151                    proof: None,
152                    transactions: Vec::new(),
153                })
154            }
155        }
156        .boxed()
157    })?;
158
159    Ok(api)
160}

177pub(super) fn node<S>() -> Result<Api<S, node::Error, StaticVersion<0, 1>>>
178where
179    S: 'static + Send + Sync + ReadState,
180    <S as ReadState>::State:
181        Send + Sync + StakeTableDataSource<SeqTypes> + NodeDataSource<SeqTypes>,
182{
183    // Extend the base API
184    let mut options = node::Options::default();
185    let extension = toml::from_str(include_str!("../../api/node.toml"))?;
186    options.extensions.push(extension);
187
188    // Create the base API with our extensions
189    let mut api = node::define_api::<S, SeqTypes, _>(&options, SequencerApiVersion::instance())?;
190
191    // Tack on the application logic
192    api.at("stake_table", |req, state| {
193        async move {
194            // Try to get the epoch from the request. If this fails, error
195            // as it was probably a mistake
196            let epoch = EpochNumber::new(req.integer_param("epoch_number").map_err(|_| {
197                hotshot_query_service::node::Error::Custom {
198                    message: "Epoch number is required".to_string(),
199                    status: StatusCode::BAD_REQUEST,
200                }
201            })?);
202
203            Ok(state
204                .read(|state| state.get_stake_table(Some(epoch)).boxed())
205                .await)
206        }
207        .boxed()
208    })?
209    .at("stake_table_current", |_, state| {
210        async move {
211            Ok(state
212                .read(|state| state.get_stake_table(None).boxed())
213                .await)
214        }
215        .boxed()
216    })?;
217
218    Ok(api)
219}
220pub(super) fn submit<N, P, S, ApiVer: StaticVersionType + 'static>() -> Result<Api<S, Error, ApiVer>>
221where
222    N: ConnectedNetwork<PubKey>,
223    S: 'static + Send + Sync + ReadState,
224    P: SequencerPersistence,
225    S::State: Send + Sync + SubmitDataSource<N, P>,
226{
227    let toml = toml::from_str::<toml::Value>(include_str!("../../api/submit.toml"))?;
228    let mut api = Api::<S, Error, ApiVer>::new(toml)?;
229
230    api.at("submit", |req, state| {
231        async move {
232            let tx = req
233                .body_auto::<Transaction, ApiVer>(ApiVer::instance())
234                .map_err(Error::from_request_error)?;
235
236            let hash = tx.commit();
237            state
238                .read(|state| state.submit(tx).boxed())
239                .await
240                .map_err(|err| Error::internal(err.to_string()))?;
241            Ok(hash)
242        }
243        .boxed()
244    })?;
245
246    Ok(api)
247}

279pub(super) fn catchup<S, ApiVer: StaticVersionType + 'static>(
280    _: ApiVer,
281) -> Result<Api<S, Error, ApiVer>>
282where
283    S: 'static + Send + Sync + ReadState,
284    S::State: Send + Sync + NodeStateDataSource + CatchupDataSource,
285{
286    let toml = toml::from_str::<toml::Value>(include_str!("../../api/catchup.toml"))?;
287    let mut api = Api::<S, Error, ApiVer>::new(toml)?;
288
289    api.get("account", |req, state| {
290        async move {
291            let height = req
292                .integer_param("height")
293                .map_err(Error::from_request_error)?;
294            let view = req
295                .integer_param("view")
296                .map_err(Error::from_request_error)?;
297            let account = req
298                .string_param("address")
299                .map_err(Error::from_request_error)?;
300            let account = account.parse().map_err(|err| {
301                Error::catch_all(
302                    StatusCode::BAD_REQUEST,
303                    format!("malformed account {account}: {err}"),
304                )
305            })?;
306
307            state
308                .get_account(
309                    state.node_state().await,
310                    height,
311                    ViewNumber::new(view),
312                    account,
313                )
314                .await
315                .map_err(|err| Error::catch_all(StatusCode::NOT_FOUND, format!("{err:#}")))
316        }
317        .boxed()
318    })?
319    .at("accounts", |req, state| {
320        async move {
321            let height = req
322                .integer_param("height")
323                .map_err(Error::from_request_error)?;
324            let view = req
325                .integer_param("view")
326                .map_err(Error::from_request_error)?;
327            let accounts = req
328                .body_auto::<Vec<FeeAccount>, ApiVer>(ApiVer::instance())
329                .map_err(Error::from_request_error)?;
330
331            state
332                .read(|state| {
333                    async move {
334                        state
335                            .get_accounts(
336                                state.node_state().await,
337                                height,
338                                ViewNumber::new(view),
339                                &accounts,
340                            )
341                            .await
342                            .map_err(|err| {
343                                Error::catch_all(StatusCode::NOT_FOUND, format!("{err:#}"))
344                            })
345                    }
346                    .boxed()
347                })
348                .await
349        }
350        .boxed()
351    })?
352    .get("blocks", |req, state| {
353        async move {
354            let height = req
355                .integer_param("height")
356                .map_err(Error::from_request_error)?;
357            let view = req
358                .integer_param("view")
359                .map_err(Error::from_request_error)?;
360
361            state
362                .get_frontier(state.node_state().await, height, ViewNumber::new(view))
363                .await
364                .map_err(|err| Error::catch_all(StatusCode::NOT_FOUND, format!("{err:#}")))
365        }
366        .boxed()
367    })?
368    .get("chainconfig", |req, state| {
369        async move {
370            let commitment = req
371                .blob_param("commitment")
372                .map_err(Error::from_request_error)?;
373
374            state
375                .get_chain_config(commitment)
376                .await
377                .map_err(|err| Error::catch_all(StatusCode::NOT_FOUND, format!("{err:#}")))
378        }
379        .boxed()
380    })?;
381
382    Ok(api)
383}

410pub(super) fn config<S, ApiVer: StaticVersionType + 'static>(
411    _: ApiVer,
412) -> Result<Api<S, Error, ApiVer>>
413where
414    S: 'static + Send + Sync + ReadState,
415    S::State: Send + Sync + HotShotConfigDataSource,
416{
417    let toml = toml::from_str::<toml::Value>(include_str!("../../api/config.toml"))?;
418    let mut api = Api::<S, Error, ApiVer>::new(toml)?;
419
420    let env_variables = get_public_env_vars()
421        .map_err(|err| Error::catch_all(StatusCode::INTERNAL_SERVER_ERROR, format!("{err:#}")))?;
422
423    api.get("hotshot", |_, state| {
424        async move { Ok(state.get_config().await) }.boxed()
425    })?
426    .get("env", move |_, _| {
427        {
428            let env_variables = env_variables.clone();
429            async move { Ok(env_variables) }
430        }
431        .boxed()
432    })?;
433
434    Ok(api)
435}

437fn get_public_env_vars() -> Result<Vec<String>> {
438    let toml: toml::Value = toml::from_str(include_str!("../../api/public-env-vars.toml"))?;
439
440    let keys = toml
441        .get("variables")
442        .ok_or_else(|| toml::de::Error::custom("variables not found"))?
443        .as_array()
444        .ok_or_else(|| toml::de::Error::custom("variables is not an array"))?
445        .clone()
446        .into_iter()
447        .map(|v| v.try_into())
448        .collect::<Result<BTreeSet<String>, toml::de::Error>>()?;
449
450    let hashmap: HashMap<String, String> = env::vars().collect();
451    let mut public_env_vars: Vec<String> = Vec::new();
452    for key in keys {
453        let value = hashmap.get(&key).cloned().unwrap_or_default();
454        public_env_vars.push(format!("{key}={value}"));
455    }
456
457    Ok(public_env_vars)
